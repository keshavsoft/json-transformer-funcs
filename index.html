Minimal Quickstart Snippet

      

      

        Copy and paste this into any ES module project or CDN-powered web app:
      

      
<!-- Load Bootstrap for styling -->
<link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.3/dist/css/bootstrap.min.css" rel="stylesheet">

<!-- Mount Point -->
<div id="my-table" class="table-responsive"></div>

<script type="module">
  // 1. Import transformer function & json-to-tag from CDN
  import { tableFromData } from "https://cdn.jsdelivr.net/npm/json-transformer-funcs/src/index.js";
  import buildSpecElement from "https://keshavsoft.github.io/json-to-tag/dist/v9/min.js";

  // 2. Define data, cell blueprint, and table scaffold
  const data = {
    columns: [{ columnName: "Product" }, { columnName: "Price" }],
    rows: [{ product: "Drill Bit", price: "$124.50" }]
  };

  const spec = {
    buildHeader: { tagName: "th", textContent: "${columnName}" },
    buildBody: {
      outerTemplate: { tagName: "tr" },
      innerTemplate: { tagName: "td", textContent: "${rowValue}" }
    }
  };

  const table = {
    tagName: "table",
    attributes: { class: "table table-hover table-striped" },
    children: [
      { tagName: "thead", children: [{ tagName: "tr", children: [] }] },
      { tagName: "tbody", children: [] }
    ]
  };

  // 3. Compile Spec tree and render native DOM!
  const compiledSpec = tableFromData({ inData: data, inSpec: spec, inTable: table });
  const liveDom = buildSpecElement(compiledSpec);

  document.getElementById("my-table").appendChild(liveDom);
</script>
