        How This Example Works
      

      
// 1. Import transformer and json-to-tag from CDN
import { tableFromData } from "json-transformer-funcs";
import buildSpecElement from "https://keshavsoft.github.io/json-to-tag/dist/v9/min.js";

// 2. Load the 3 distinct JSON inputs
import data from "./data.json" with { type: "json" };
import spec from "./spec.json" with { type: "json" };
import table from "./table.json" with { type: "json" };

// 3. Compile raw data & templates into full Table Spec Tree
const compiledSpec = tableFromData({
  inData: data,
  inSpec: spec,
  inTable: table
});

// 4. Render native HTML table DOM via json-to-tag CDN
const domElement = buildSpecElement(compiledSpec);

// 5. Mount into DOM container
document.getElementById("table-container").appendChild(domElement);
